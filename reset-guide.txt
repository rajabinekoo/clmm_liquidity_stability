DELETE FROM lp_actions WHERE block_number >= 16675015;

SET

        SELECT MAX(block_number)
        FROM pool_snapshots
        WHERE pool_address =
              '0x88e6a0c2ddd26feeb64f039a2c41296fcb3f5640'

WHERE pool_address =
      '0x88e6a0c2ddd26feeb64f039a2c41296fcb3f5640'
